# =====================================================================
# Deep Doc Search — зависимости окружения разработки.
#
# Назначение:
#   Единый файл для установки всех зависимостей, необходимых для
#   локальной разработки, тестирования, статического анализа,
#   бенчмаркинга и запуска smoke-тестов. Runtime-зависимости
#   включены, чтобы разработчик не устанавливал отдельно
#   requirements.txt и requirements-dev.txt.
#
# Установка:
#   python -m venv .venv
#   source .venv/bin/activate
#   pip install -r requirements-dev.txt
#
# После установки для smoke-тестов требуется браузерный движок:
#   playwright install chromium
#
# Группировка:
#   1. Runtime         — библиотеки, необходимые для запуска DDS.
#   2. Testing         — pytest и его расширения.
#   3. Static analysis — линтеры и проверки архитектуры.
#   4. Benchmarking    — замер производительности.
#   5. Smoke tests     — браузерные тесты UI.
#
# Совместимость: Python 3.12+ (см. docs/deployment.md).
# =====================================================================


# ---------------------------------------------------------------------
# 1. Runtime
#
# Эти зависимости также продублированы в requirements.txt.
# Дублирование осознанное: developer-окружение должно быть
# самодостаточным — одна команда pip install и приложение запускается.
# ---------------------------------------------------------------------

# Извлечение текста, рендер страниц, построение индексов слов.
pymupdf>=1.23.0

# Экранирование в Jinja2 (транзитивная зависимость, явно фиксируем
# минимальную версию с исправлениями CVE-2024-22195).
markupsafe>=2.1.0

# Веб-фреймворк: REST API, dependency injection, lifespan.
fastapi>=0.100.0

# ASGI-сервер. [standard] включает uvloop и httptools —
# значительно ускоряет event loop на Linux.
uvicorn[standard]>=0.23.0

# Шаблонизатор для серверного рендеринга HTML-страниц.
jinja2>=3.1.0

# Валидация запросов/ответов API. v2 обязательна —
# используются field_validator и совместимость с FastAPI ≥ 0.100.
pydantic>=2.0.0


# ---------------------------------------------------------------------
# 2. Testing
# ---------------------------------------------------------------------

# Базовый фреймворк.
pytest>=7.0.0

# Поддержка async-тестов. Критично для тестирования ScanPipeline,
# ProcessTaskRunner и других корутин.
pytest-asyncio>=0.21.0

# Измерение покрытия тестами. Порог ≥ 60% для критических модулей
# (см. критерии Фазы 0 в плане рефакторинга).
pytest-cov>=4.1.0


# ---------------------------------------------------------------------
# 3. Static analysis
#
# Задачи:
#   - Проверка направления зависимостей между слоями (domain,
#     application, infrastructure, dds_web).
#   - Проверка отсутствия SQL-строк в контрактах application-слоя.
#   - Статическая типизация.
#   - Форматирование и линтинг.
# ---------------------------------------------------------------------

# Проверка архитектурных контрактов: запрет импортов между слоями
# в неверном направлении. Конфигурация генерируется скриптом
# tools/generate_importlinter_config.py из docs/architecture-decisions/exceptions.yaml.
import-linter>=2.0

# Статическая типизация. Критично для контроля протечек типов
# через контракты (например, tuple vs list в SearchResult.pages).
mypy>=1.13.0

# Линтер и форматтер. Заменяет flake8 + isort + black.
ruff>=0.1.0

# Хуки для pre-commit (см. .pre-commit-config.yaml).
pre-commit>=3.5.0


# ---------------------------------------------------------------------
# 4. Benchmarking
#
# Задачи:
#   - Сравнение производительности до и после фаз рефакторинга.
#   - Статистическая значимость различий (Mann-Whitney U).
# ---------------------------------------------------------------------

# Замеры времени выполнения, интеграция с pytest.
# Позволяет запускать бенчмарки как обычные тесты.
pytest-benchmark>=4.0.0

# Статистический анализ: Mann-Whitney U test для сравнения
# медиан (непараметрический тест, устойчивый к выбросам).
scipy>=1.11.0


# ---------------------------------------------------------------------
# 5. Smoke tests
#
# Задачи:
#   - Проверка работоспособности UI после каждой фазы:
#     поиск, скачивание, рендер, подсветка, настройки.
#   - Прогон в CI-job smoke (не блокирует merge).
# ---------------------------------------------------------------------

# Браузерные тесты. После установки требуется:
#   playwright install chromium
playwright>=1.40.0


# ---------------------------------------------------------------------
# 6. Вспомогательное
# ---------------------------------------------------------------------

# Парсинг YAML-конфигов: exceptions.yaml (источник истины для
# .importlinter) и DEPRECATIONS.yaml (учёт устаревших символов).
pyyaml>=6.0
