# =====================================================================
# Deep Doc Search — runtime-зависимости.
#
# Назначение:
#   Минимальный набор библиотек, необходимых для запуска DDS
#   в production. Не содержит инструментов разработки (pytest,
#   ruff, mypy, playwright и т. п.).
#
#   Развёртывание:
#     python -m venv .venv
#     source .venv/bin/activate      # или activate.fish
#     pip install -r requirements.txt
#     cp config.json.example config.json
#     # отредактировать config.json
#     python run.py --config config.json
#
#   Для локальной разработки (с тестами и линтерами) использовать
#   requirements-dev.txt, который включает этот файл через
#   директиву -r и добавляет dev-инструменты.
#
# Единственный источник runtime-зависимостей:
#   Изменения runtime-пакетов (версии, добавление, удаление)
#   вносятся ТОЛЬКО здесь. requirements-dev.txt подтягивает
#   этот файл через -r requirements.txt и не дублирует содержимое.
#
# Группировка:
#   1. Веб-фреймворк.
#   2. Валидация.
#   3. Шаблонизация.
#   4. Обработка форм.
#   5. PDF-обработка.
#
# Совместимость: Python 3.12+ (см. docs/deployment.md).
# =====================================================================


# ---------------------------------------------------------------------
# 1. Веб-фреймворк
# ---------------------------------------------------------------------

# REST API, dependency injection, lifespan.
fastapi>=0.100.0

# ASGI-сервер. [standard] включает uvloop и httptools —
# значительно ускоряет event loop на Linux.
uvicorn[standard]>=0.23.0


# ---------------------------------------------------------------------
# 2. Валидация
# ---------------------------------------------------------------------

# Валидация запросов/ответов API. v2 обязательна —
# используются field_validator и совместимость с FastAPI ≥ 0.100.
pydantic>=2.0.0


# ---------------------------------------------------------------------
# 3. Шаблонизация
# ---------------------------------------------------------------------

# Шаблонизатор для серверного рендеринга HTML-страниц.
jinja2>=3.1.0

# Экранирование в Jinja2 (транзитивная зависимость, явно фиксируем
# минимальную версию с исправлениями CVE-2024-22195).
markupsafe>=2.1.0


# ---------------------------------------------------------------------
# 4. Обработка форм
# ---------------------------------------------------------------------

# Обработка multipart/form-data. Требуется FastAPI для Form(...).
# Используется в эндпоинте POST /login (dds_web/pages.py).
# Без этого пакета приложение падает при импорте pages.py с
# ошибкой: Form data requires "python-multipart" to be installed.
# Добавлен явно: в CI не устанавливается транзитивно.
python-multipart>=0.0.6


# ---------------------------------------------------------------------
# 5. PDF-обработка
# ---------------------------------------------------------------------

# Извлечение текста, рендер страниц, построение индексов слов.
pymupdf>=1.23.0
